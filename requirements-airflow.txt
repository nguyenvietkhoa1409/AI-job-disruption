# structural test (tests/test_dag_integrity.py), which needs nothing
# version-specific, so let the constraint pick the version.
--constraint https://raw.githubusercontent.com/apache/airflow/constraints-2.10.4/constraints-3.12.txt
apache-airflow==2.10.4
pytest
